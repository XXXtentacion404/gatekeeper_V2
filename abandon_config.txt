# ============================================================================
# 物品滞留检测配置（纯 CV 背景差分 · 不识别物品类别）
# ----------------------------------------------------------------------------
# 独立于 cameras_config.txt —— 刻意不把参数写进相机行，避免动到已验收的计数配置。
# 文件不存在时全部用内置默认值（不会报错）。
# 语法：key=value，# 或 ; 开头为注释。
# 按相机角色覆盖：role0.xxx / role1.xxx（0=入口相机, 1=出口相机）。
#
# 详细说明见 文档/物品滞留检测_模块设计.md
# ============================================================================

# ---- 总开关（0 = 完全关闭，主循环里 update() 立即返回）----
enable=1

# ---- 背景差分工作分辨率与抽样 ----
# 越小越快、越小目标越糊。960x540 时 8cm 目标 ≈25px（见设计文档 §8）。
# sample_every=4 → 每 4 帧跑一次差分（20FPS 下等效 5fps），CPU ↓4×。
# 滞留是秒级事件，抽样完全够；这是「不占 NPU、CPU 只 ~4-5%」的关键。
work_w=960
work_h=540
warmup_frames=60
sample_every=4
use_own_thread=0

# ---- 可选：标准空场景背景图（导师"两输入"的输入①）----
# 填了就直接用这张图 seed 长/短背景 → 免 warmup 冷启动，开机即检测。
# 留空 = 靠 warmup_frames 冷启动。建议路径（板端）：
#   bg_seed_path=/home/topeet/code/gatekeeper_V4/abandon_bg.jpg
bg_seed_path=

# ---- 双背景学习率（决定有效检测窗口，改前先读这段）----
# 有效窗口 ≈ [ 1/bg_short_lr , 1/bg_long_lr ] 个抽样帧：
#   bg_short_lr=0.05  → 静止物约 20 个抽样帧后被快背景吸收（≈4s @4帧抽样）
#   bg_long_lr =0.002 → 静止物约 500 个抽样帧后也被慢背景吸收（≈100s）
# ⚠️ push_sec 必须 < 窗口上限，否则滞留物会从 static_mask 消失、二级告警发不出来。
#    要更长的门槛（如 180s）就同步调小 bg_long_lr（如 0.001）。
bg_long_lr=0.002
bg_short_lr=0.05

# ---- 尺寸门限（硬指标：最小 8cm×8cm）----
# min_item_px 是「全分辨率像素」口径，需现场按 px_per_cm 标定：
#   px_per_cm = 图像宽(px) / 视野宽(cm)；8cm ≈ 8 × px_per_cm。
# 1920 宽、检测区视野约 3m(300cm) → px_per_cm≈6.4 → 8cm≈51px（现在的 45 是留裕度值）。
# 用已知尺寸物体（A4 纸 21×29.7cm）在两个距离各测一次即可反推 px_per_cm。
min_item_px=45
min_item_area=2000
max_area_ratio=0.03

# ---- 静止块跟踪 ----
# assoc_dist_px：相邻抽样帧质心关联距离（全分辨率）
# reset_dist_px：帧间位移超此值 = "被移动" → 计时清零（对应"移动位置，重置计时"）
# item_lost_frames：连续丢失多少个抽样帧才算"被拾取"（容忍行人短暂遮挡）
assoc_dist_px=60
item_lost_frames=10
reset_dist_px=10
smooth_alpha=0.5

# ---- 人-物绑定 ----
# safe_radius_px：人框中心 ↔ 物品质心 的"看管"半径。半径内有人 → 计时暂停。
# person_iou_reject：静止块与 person 框 IoU 超此值 → 判为"站着不动的人"，丢弃。
safe_radius_px=120
bind_confirm_frames=5
person_iou_reject=0.30

# ---- 两级告警（秒）----
# remind_sec：一级，提醒乘客；push_sec：二级，推送监控中心/保安（硬指标默认 60s）。
# 只想要单一 60s 门槛 → 把 remind_sec 也设成 60。
remind_sec=15
push_sec=60

# ---- 证据序列（滞留前的人-物关联图片序列）----
# 有人看管期间按 seq_fps 记录小图，主人离开那一刻把最近 backtrack_sec 秒落盘。
backtrack_sec=10
seq_fps=5
save_dir=./abandon_evidence
