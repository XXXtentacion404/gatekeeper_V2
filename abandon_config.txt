# ============================================================================
# 物品滞留检测配置 —— 清场双时相版（2026-10-20 整块重写）

# 旧版（双 MOG2 / 参考背景差分 / 人-物绑定）的键已全部废弃：mode、ref_bg_path、
# bg_long_lr、bg_short_lr、safe_radius_px、bind_confirm_frames、person_iou_reject、
# reset_dist_px、assoc_dist_px、item_lost_frames、ref_diff_thresh、warmup_frames …
# 读进来会被静默忽略（不报错），所以本文件里不再保留它们。
# 旧实现完整代码见 commit c67cd6a。
#
# 新思路：清场触发 → 三帧比对（长期基线 / 入场前 / 离场后）→ 变化块 →
#         几何配对（判"被移动"）→ 残留表跨会话确认 → 两级告警。
# 详细设计见 文档/FastBSUV-Net2.0_清场双时相比对方案.md
# ============================================================================

# ---- 总开关（0 = update() 立即返回，零 NPU/CPU 开销）----
# ⚠️ 当前保持 0：现场还在跑「进出人数」计数基线，先不让滞留检测搅局。
#    ★ 要验证本模块时改成 1，不用重新编译，改完重跑即可（启动会打印 [ITEM] 参数行）。
enable=0

# ============================================================================
# 一、时序（秒）—— 决定"什么时候才允许比一次"
# ----------------------------------------------------------------------------
# t_clear_sec   连续无人多久 → 才算"确认清场"。太短会把"人刚走过去"误当清场，
#               拍到的 before 帧里还有残影；1.5~2.0 是现场验过的量级。
# t_settle_sec  人走光后等多久才取 after 帧。人刚离开时地面还有影子、曝光还在回调，
#               立刻取帧会把这两样当成"新增块"。2.0s 起步，逆光场景可加到 3.0。
# max_occupied_sec  有人持续待超此时长 → 放弃本次会话（否则一直等不到清场，
#               状态机永远卡在 OCCUPIED，后面的场次全丢）。
# ============================================================================
t_clear_sec=1.5
t_settle_sec=2.0
max_occupied_sec=300

# ============================================================================
# 二、检测区裁剪（★ 8cm 小目标能不能做到的关键）
# ----------------------------------------------------------------------------
# 只把「检测区多边形」的 bbox 送进比对，且 1:1 不降采样。
# 整帧 1920x1080 缩到 960x540 会让 8cm 目标像素数直接砍半 → 直接判死。
# 现场检测区约占画面 27%，裁剪后 CPU 反而更低。
# ============================================================================
use_roi_crop=1
roi_margin_px=24
work_max_w=1280      # 裁剪后宽度超此值才降采样；0 = 永不降采样
work_w=960           # 仅 use_roi_crop=0（整帧模式）时用
work_h=540

# ============================================================================
# 三、标定（决定 8cm 指标是否达成）
# ----------------------------------------------------------------------------
# region_width_cm = 检测区横向的实际物理宽度(cm)。填了启动时会打印 px_per_cm
# 和 8cm 对应多少像素，并直接告警"8cm 会不会被尺寸门限滤掉"。
# 测法：检测区里横放一张 A4 纸(29.7cm)，量它在画面里的像素宽，px_per_cm = 像素宽/29.7。
#   经验门槛：8cm 边长 ≥20px 可靠 | 15~20px 勉强 | <15px 不可靠
# ============================================================================
region_width_cm=0    # ⚠️ 还没标定，现场量一次填上（填了才能自检 8cm）

# ---- 尺寸门限（全分辨率口径）----
# min_item_area：8cm ≈ 18px 见方 ≈ 324px²。旧值 2000 会把 8cm 直接滤掉。
# min_item_px  ：最小边长。与 min_item_area 必须同步改，一个管边长一个管面积。
# max_area_ratio：单块面积占处理图比例上限，防"整屏变化"被当成一个巨块。
max_area_ratio=0.03
min_item_px=45
min_item_area=324

# ============================================================================
# 四、变化掩膜（当前 = 纯 CV 骨架，将来换成 Fast BSUV-Net 2.0）
# ----------------------------------------------------------------------------
# change_thresh  |before − after| 灰度阈值。越小越敏感、越能抓小目标，噪声也越多。
#               8cm 场景 20~25；目标像素 <20px 时可压到 15。
# morph_open_px  开运算：吃细碎噪点，但也会吃细结构。目标 <20px 时用 0 或 3。
# morph_close_px 闭运算：补小目标断裂（阈值高时很容易断成两半）。
# ============================================================================
change_thresh=25
morph_open_px=3
morph_close_px=5

# ---- 推理后端 ----
# use_network=0 走纯 CV（absdiff）；=1 走 RKNN 网络 —— 目前只接了占位符，
# 设 1 会自动退化回 cv 并在日志里注明 network-not-ready。
# 唯一要改的地方是 src/item_abandonment.cpp 的 buildChangeMask()。
use_network=0
model_path=

# ============================================================================
# 五、会话作废守卫（宁可漏一次，也不要拿两张不可比的图比出满屏候选）
# ----------------------------------------------------------------------------
# max_mean_abs_diff  before/after 全图平均绝对差(灰度级)。超此值基本就是光照变了。
# max_changed_ratio  变化像素占比上限。超此值 = 相机被撞歪 / 有人挡镜头 / 开关灯。
# ============================================================================
max_mean_abs_diff=18
max_changed_ratio=0.25

# ============================================================================
# 六、判"有没有人"
# ----------------------------------------------------------------------------
# motion_thresh  ROI 内相邻帧平均灰度差。这是 YOLO 之外的兜底：
#               YOLO 漏掉一个蹲着的人时，只看框数会误判"清场了"，
#               然后那个人自己就被报成遗留物。
# ============================================================================
motion_thresh=6

# ============================================================================
# 七、正负配对（"物体被挪了位置" ≠ "遗留"）
# ----------------------------------------------------------------------------
# 一个东西从 A 挪到 B，差分出两块对称的块：A 处"消失"、B 处"新增"。
# 必须纯几何配对（面积相近 + 质心相近）先吃掉，否则会被当成"新放了一个"。
# pair_dist_px    两块质心距离上限。ROI 宽约 1100px → 100~150 合理，60 偏小。
# pair_area_ratio 面积比下限（小/大）。透视角变化会让同一物体在画面上下面积不同，
#                 0.5 是留了余量；要求等大(1.0)会配不上。
# ============================================================================
pair_dist_px=120
pair_area_ratio=0.5

# ============================================================================
# 八、残留表（跨会话确认）
# ----------------------------------------------------------------------------
# 一块新增块只比一次不算数 —— 下一轮清场如果它还在，才算"真留下了"。
# confirm_rounds       连续 N 轮清场都稳定存在 → 确认，可报警。
# keep_missed_rounds   连续 N 轮没匹配上 → 认为被取走，从残留表清除。
# ⚠️ 残留物所在区域会被排除在基线更新之外（否则它会被吸进基线，之后永远检不出来）。
# ============================================================================
confirm_rounds=2
keep_missed_rounds=2

# ============================================================================
# 九、两级告警（秒，从首次发现那一刻起算）
# ----------------------------------------------------------------------------
# remind_sec 一级：现场提示；push_sec 二级：推送监控中心（硬指标 60s）。
# 只想要单一 60s 门槛 → 把 remind_sec 也设成 60。
# ⚠️ 调试点先设 15/60，别学旧版设 5 —— 5s 时路人从包旁走过就会触发。
# ============================================================================
remind_sec=15
push_sec=60

# ============================================================================
# 十、兼容字段（读进来但当前版本不实现）
# ----------------------------------------------------------------------------
# 证据序列（滞留前的人-物关联图）在新架构里不做了：清场双时相本身就把"谁放的"
# 变成了"无人期间检测到的变化"，不再需要往回翻帧。留着键防止旧脚本报错。
# ============================================================================
save_dir=./abandon_evidence
backtrack_sec=10
seq_fps=5
