# ============================================================================
# 物品滞留检测配置（纯 CV 背景差分 · 不识别物品类别）
# ----------------------------------------------------------------------------
# 独立于 cameras_config.txt —— 刻意不把参数写进相机行，避免动到已验收的计数配置。
# 文件不存在时全部用内置默认值（不会报错）。
# 语法：key=value，# 或 ; 开头为注释。
# 按相机角色覆盖：role0.xxx / role1.xxx（0=入口相机, 1=出口相机）。
#
# 详细说明见 文档/物品滞留检测_模块设计.md
# ============================================================================

# ---- 总开关（0 = 完全关闭，主循环里 update() 立即返回）----
enable=1

# ============================================================================
# 一、检测模式 —— 决定「什么算滞留物」
#   mog2   : 双自适应背景 fg_long & ~fg_short
#            · 抗光照变化强（自动学习），但小目标(8cm)弱（统计阈值难触发）
#            · 有吸收窗口上限 ≈ 1/bg_long_lr 个抽样帧（默认约 100s），push_sec 不能超
#   ref    : 参考背景差分 |当前帧 − 空场景标准图| > 阈值   ← 导师"两输入"方案
#            · 对小目标最敏感（阈值可压到 15 灰阶），· 但光照一变整屏误报、常驻新物持续误报
#   hybrid : ★推荐 = 参考差分(给灵敏度) 与 短背景非前景(给"已静止"语义)
#            · 既抓得住 8cm，又能压掉走动的人；配合下面的光照突变守卫
# ⚠️ ref / hybrid 必须先填 ref_bg_path，否则启动时自动退回 mog2 并告警。
# ============================================================================
mode=hybrid

# ---- 标准空场景背景图（导师"两输入"的输入①：画面里没有任何滞留物、无行人）----
# 作用：① 参考差分的基线  ② 用来 seed 双 MOG2 → 跳过 warmup，开机即检测。
# 板端建议路径： /home/topeet/code/gatekeeper_V4/abandon_bg.jpg
# 拍摄要求：与运行时同一机位、同一光照时段；人走空后连拍取中值更好（见下方说明）。
# 留空 = 退回 mog2 冷启动（warmup_frames 帧后开始检测）。
ref_bg_path=

# ============================================================================
# 二、检测区裁剪（★ 8cm 小目标能不能做到的关键）
#   只把「检测区多边形」的 bbox 送进差分，并尽量保持 1:1 分辨率，
#   而不是把整帧 1920x1080 缩到 960x540 —— 后者会让 8cm 目标像素数直接砍半。
#   现场检测区约占画面 27%，裁剪后 CPU 反而更低。
# ============================================================================
use_roi_crop=1
roi_margin_px=24
work_max_w=1280

# ============================================================================
# 三、标定（决定 8cm 指标是否达成，强烈建议填）
#   region_width_cm = 检测区横向的实际物理宽度(cm)。
#   填了启动时会打印：px_per_cm、8cm 对应多少像素、min_item_px 折算多少 cm，
#   并直接告警"8cm 会不会被尺寸门限滤掉"或"像素太少不可靠"。
#   现场测法：在检测区放 A4 纸横放(29.7cm)，量它在画面里的像素宽，px_per_cm = 像素宽/29.7。
# 经验门槛（处理分辨率下 8cm 的边长）：
#   ≥20px 可靠 | 15~20px 勉强(建议 morph_open_px=0 + ref_diff_thresh 降到 15~20) | <15px 不可靠
# ============================================================================
region_width_cm=0

# ---- 尺寸门限（硬指标：最小 8cm×8cm）----
# min_item_px 是「全分辨率像素」口径。按上面 region_width_cm 标定后填 8×px_per_cm
# （不要盲目沿用 45，先用标定自检打印的值）。
min_item_px=45
min_item_area=2000
max_area_ratio=0.03

# ---- 形态学（小目标关键：开运算会吃掉细结构）----
# morph_open_px=0 表示不做开运算（目标 <20px 时推荐 0 或 3）。
# 原默认 5/9 是给大目标去噪的，对 8cm 太狠。
morph_open_px=3
morph_close_px=5

# ============================================================================
# 四、参考差分参数（mode=ref / hybrid 时生效）
#   ref_diff_thresh：灰度差阈值。越小越敏感、越能抓小目标，但噪声误报增多。
#                    8cm 场景建议 15~25；目标像素 <20px 时可压到 15。
#   光照守卫：一帧里差分前景占比 > ref_light_change_ratio → 判定为光照/相机突变，
#            本帧不出候选；连续 ref_light_change_frames 帧如此 → 用当前帧刷新参考基线
#            （并打印告警；若此刻画面里真有滞留物，它会被写进基线而不再告警，需人工确认）。
# ============================================================================
ref_diff_thresh=25
ref_light_change_ratio=0.25
ref_light_change_frames=3

# ============================================================================
# 五、通用参数
# ============================================================================
# ---- 抽样与工作分辨率 ----
# sample_every=4 → 每 4 帧跑一次差分（20FPS 下等效 5fps），CPU ↓4×。
# 滞留是秒级事件，抽样完全够；这是「不占 NPU、CPU 只几个百分点」的关键。
sample_every=4
use_own_thread=0
# 仅 use_roi_crop=0（整帧模式）或 ROI 宽度超 work_max_w 时才用到下面两个
work_w=960
work_h=540
warmup_frames=60

# ---- 双背景学习率（决定 mog2 模式的有效检测窗口，改前先读这段）----
# 有效窗口 ≈ [ 1/bg_short_lr , 1/bg_long_lr ] 个抽样帧：
#   bg_short_lr=0.05  → 静止物约 20 个抽样帧后被快背景吸收（≈4s @4帧抽样）
#   bg_long_lr =0.002 → 静止物约 500 个抽样帧后也被慢背景吸收（≈100s）
# ⚠️ mog2 模式下 push_sec 必须 < 窗口上限。要更长门槛就调小 bg_long_lr，
#    或直接用 ref/hybrid 模式（无此上限）。
# hybrid 模式里短背景只负责"是否已静止"，此参数仍起作用。
bg_long_lr=0.002
bg_short_lr=0.05

# ---- 静止块跟踪 ----
# assoc_dist_px：相邻抽样帧质心关联距离（全分辨率）
# reset_dist_px：帧间位移超此值 = "被移动" → 计时清零（对应"移动位置，重置计时"）
# item_lost_frames：连续丢失多少个抽样帧才算"被拾取"（容忍行人短暂遮挡）
assoc_dist_px=60
item_lost_frames=10
reset_dist_px=10
smooth_alpha=0.5

# ---- 人-物绑定 ----
# safe_radius_px：人框中心 ↔ 物品质心 的"看管"半径。半径内有人 → 计时暂停。
# person_iou_reject：静止块与 person 框 IoU 超此值 → 判为"站着不动的人"，丢弃。
safe_radius_px=120
bind_confirm_frames=5
person_iou_reject=0.30

# ---- 两级告警（秒）----
# remind_sec：一级，提醒乘客；push_sec：二级，推送监控中心/保安（硬指标默认 60s）。
# 只想要单一 60s 门槛 → 把 remind_sec 也设成 60。
remind_sec=15
push_sec=60

# ---- 证据序列（滞留前的人-物关联图片序列）----
# 有人看管期间按 seq_fps 记录小图，主人离开那一刻把最近 backtrack_sec 秒落盘。
backtrack_sec=10
seq_fps=5
save_dir=./abandon_evidence
